cmake_minimum_required(VERSION 3.16)
project(mower_nav_navfn_oracle CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav2_navfn_planner REQUIRED)
find_package(nav2_util REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(oracle oracle.cpp)
ament_target_dependencies(oracle rclcpp rclcpp_lifecycle nav2_core nav2_costmap_2d nav2_navfn_planner
  nav2_util tf2 tf2_ros geometry_msgs nav_msgs)
target_link_libraries(oracle nlohmann_json::nlohmann_json)
